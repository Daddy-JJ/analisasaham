// Helper for Shape Mapping
f_get_shape(_s) =>
    switch _s
        "Triangle Up" => shape.triangleup
        "Triangle Down" => shape.triangledown
        "Circle" => shape.circle
        "Square" => shape.square
        "Diamond" => shape.diamond
        "Label Up" => shape.labelup
        "Label Down" => shape.labeldown
        => shape.circle

// --- IDX PRICE FRACTION LOGIC ---
f_tick_idx(_p) =>
    float _s = 1.0
    if _p < 200
        _s := 1.0
    else if _p < 500
        _s := 2.0
    else if _p < 2000
        _s := 5.0
    else if _p < 5000
        _s := 10.0
    else
        _s := 25.0
    math.round(_p / _s) * _s

// --- SMART RESISTANCE HELPERS ---
f_count_touches(_lvl) =>
    int _cnt = 0
    if not na(_lvl)
        float _tol = _lvl * 0.005
        for j = 1 to 100
            if high[j] >= (_lvl - _tol) and high[j] <= (_lvl + _tol)
                _cnt += 1
    _cnt

f_get_prob(_cnt) =>
    if _cnt <= 1
        "Fresh (Hard)"
    else if _cnt <= 3
        "Weak (Hi-Prob)"
    else
        "Crit (Break?)"

type resistInfo
    float price
    int tests
    string prob_txt

f_get_smart_resistance(_src_ph) =>
    var float[] arr_ph = array.new_float(0)
    if not na(_src_ph)
        array.push(arr_ph, _src_ph)
        if array.size(arr_ph) > 30
            array.shift(arr_ph)
    
    float[] candidates = array.new_float(0)
    if array.size(arr_ph) > 0
        for i = 0 to array.size(arr_ph) - 1
            float _val = array.get(arr_ph, i)
            if _val > (close * 1.005)
                array.push(candidates, _val)
    
    array.sort(candidates, order.ascending)
    float _r1 = array.size(candidates) > 0 ? array.get(candidates, 0) : na
    float _r2 = array.size(candidates) > 1 ? array.get(candidates, 1) : na
    
    int _t1 = f_count_touches(_r1)
    int _t2 = f_count_touches(_r2)
    
    [resistInfo.new(_r1, _t1, f_get_prob(_t1)), resistInfo.new(_r2, _t2, f_get_prob(_t2))]

f_clamp(float x, float lo, float hi) => math.max(lo, math.min(hi, x))
f_safe_div(float num, float den, float fallback) => den != 0.0 ? (num / den) : fallback
f_luma(color c) =>
    float r = color.r(c), g = color.g(c), b = color.b(c)
    (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255.0

f_mix(color c1, color c2, float t) =>
    float tt = f_clamp(t, 0.0, 1.0)
    int r = int(math.round(color.r(c1) + (color.r(c2) - color.r(c1)) * tt))
    int g = int(math.round(color.g(c1) + (color.g(c2) - color.g(c1)) * tt))
    int b = int(math.round(color.b(c1) + (color.b(c2) - color.b(c1)) * tt))
    color.rgb(r, g, b)

f_calc_quant_metrics(_src, _bench_src, _len, _rf_pct) =>
    float _ret_log = math.log(_src / _src[1])
    float _bench_ret_log = math.log(_bench_src / _bench_src[1])
    
    float _mean_daily_ret = ta.sma(_ret_log, _len)
    float _ann_ret = _mean_daily_ret * 252
    float _ann_ret_bench = ta.sma(_bench_ret_log, _len) * 252
    
    float _stdev_daily = ta.stdev(_ret_log, _len)
    float _ann_vol = _stdev_daily * math.sqrt(252)
    
    float _neg_ret = math.min(_ret_log, 0)
    float _down_var = ta.sma(math.pow(_neg_ret, 2), _len)
    float _down_dev_daily = math.sqrt(_down_var)
    float _ann_down_dev = _down_dev_daily * math.sqrt(252)
    
    float _rf = _rf_pct / 100.0
    float _excess_ret = _ann_ret - _rf
    
    float _sharpe = _ann_vol != 0 ? (_excess_ret / _ann_vol) : 0.0
    float _sortino = _ann_down_dev != 0 ? (_excess_ret / _ann_down_dev) : 0.0
    float _alpha = (_ann_ret - _ann_ret_bench) * 100
    
    [_sharpe, _sortino, _alpha, _ann_vol * 100]

f_calc_engine(_use_fvg, _agresif_mode, _col_snip, _col_beta, _col_gamm, _col_vshp, _col_earl) =>
    float _atr_eng = ta.atr(14)
    float _vol_sma_eng = ta.sma(volume, 20)
    float _hh_eng = ta.highest(high, lookback_len)
    float _ll_eng = ta.lowest(low, lookback_len)
    
    // --- [HARDCODED BACK TO DEFAULT] ---
    float _ema21 = ta.ema(close, 21)
    float _ema50 = ta.ema(close, 50)
    float _ema200 = ta.ema(close, 200)
    
    // --- [HARDCODED BACK TO DEFAULT] Supertrend Logic ---
    float _up = hl2 - (2.0 * ta.atr(10))
    float _dn = hl2 + (2.0 * ta.atr(10))
    var float _trend_up = 0.0
    var float _trend_dn = 0.0
    _trend_up := close[1] > _trend_up[1] ? math.max(_up, _trend_up[1]) : _up
    _trend_dn := close[1] < _trend_dn[1] ? math.min(_dn, _trend_dn[1]) : _dn
    var int _dir = 1
    _dir := close > _trend_dn[1] ? 1 : close < _trend_up[1] ? -1 : nz(_dir[1], 1)
    float _stop_line = _dir == 1 ? _trend_up : _trend_dn
    
    // --- [HARDCODED BACK TO DEFAULT] MACD & RSI ---
    [macdLine, signalLine, _] = ta.macd(close, 12, 26, 9)
    float _rsi = ta.rsi(close, 14)
    
    float _mid_range = (_hh_eng + _ll_eng) / 2
    bool _is_below_equilibrium = close < _mid_range
    bool _is_premium_zone = close >= _mid_range
    
    float _pl = ta.pivotlow(low, pivot_left, pivot_right)
    var float _dem_top = na
    var float _dem_bot = na
    var float _last_swing_low = na
    var float _last_swing_rsi = na
    
    if not na(_pl)
        _dem_bot := _pl
        _dem_top := high[pivot_right]
        _last_swing_low := _pl
        _last_swing_rsi := _rsi[pivot_right]
    
    float _toleransi = (high - low)
    bool _is_in_struct = close <= (_dem_top + _toleransi) and close >= _dem_bot
    if close < _dem_bot
        _is_in_struct := false
        
    float _mid_fib = _ll_eng + ((_hh_eng - _ll_eng) * 0.382)
    bool _is_in_discount = close < _mid_fib
    
    // --- [HARDCODED BACK TO DEFAULT] ADX/DMI ---
    [di_plus, di_minus, adx_val] = ta.dmi(14, 14)
    
    bool _is_crashing = (adx_val > adx_limit) and (di_minus > di_plus)
    bool _safe_from_crash = use_crash_filt ? not _is_crashing : true

    // --- FVG DETECTION ---
    bool _has_fvg = (low[0] > high[2])
    bool _fvg_valid = _use_fvg ? _has_fvg : true

    // CANDLE SHAPE
    bool _prev_red = close[1] < open[1]
    bool _curr_green = close > open
    float _mid_prev = (open[1] + close[1]) / 2
    bool _piercing = close > _mid_prev
    float _body_size = math.abs(close - open)
    float _range_size = high - low
    float _upper_wick = high - close
    float _lower_wick = open - low
    bool _decent_body = _body_size > (_range_size * 0.4)
    bool _small_upper = _upper_wick < _body_size
    bool _candle_ok = _curr_green and _decent_body and _small_upper

    // SIGNALS
    bool _trend_flip = (_dir == 1) and (_dir[1] == -1)
    bool _trend_cont = (_dir == 1)
    bool _rsi_ok = _rsi > 50
    bool _rsi_cross = ta.crossover(_rsi, 50)
    bool _f_vol = volume > (_vol_sma_eng * 1.2)
    
    // SNIPER LOGIC
    bool _valid_loc_sniper = (_is_in_struct or _is_in_discount) and _is_below_equilibrium
    bool _trigger_sniper = (_trend_flip and _rsi_ok) or (_trend_cont and _rsi_cross)
    bool _sig_combo = _trigger_sniper and _valid_loc_sniper and _f_vol and _safe_from_crash and _candle_ok and _fvg_valid
    
    // BETA LOGIC
    bool _trigger_beta = _trend_flip and _rsi_ok
    bool _sig_beta = _trigger_beta and _is_premium_zone and _f_vol and _safe_from_crash and _candle_ok

    // ==========================================
    // --- REVISED V-SHAPE LOGIC (BMRI PATCH) ---
    // ==========================================
    bool _solid_body_v = _body_size > (_range_size * 0.4)
    bool _small_wick_v = _upper_wick < (_body_size * 0.8)
    bool _piercing_v = close > ((open[1] + close[1]) / 2)
    
    // Volume Filter
    bool _vol_spike_v = volume > (_vol_sma_eng * v_vol_factor)

    // 1. Classic Low
    bool _is_classic_low = low[1] <= ta.lowest(low, 10)[1]
    // 2. Delayed Low
    bool _is_delayed_low = (low[2] <= ta.lowest(low, 10)[2]) and (close[1] < open[1])
    bool _valid_bottom_loc = _is_classic_low or _is_delayed_low
    
    float _nearest_res = math.min(_ema21, _stop_line)
    float _gap_to_res = (_nearest_res - close) / close * 100
    bool _gap_ok = _dir == -1 ? (_gap_to_res > 2.0) : true
    
    bool _wick_safe = high < (_ema21 * 1.005)
    bool _cond_std = _gap_ok and _wick_safe
    
    bool _is_engulfing = close >= open[1]
    bool _cond_power = _valid_bottom_loc and _prev_red and _is_engulfing and _vol_spike_v
    
    bool _final_vshape_logic = _cond_std or _cond_power
    bool _sig_vshape = _prev_red and _curr_green and _piercing_v and _vol_spike_v and _final_vshape_logic and _safe_from_crash

    // ==========================================
    // --- GAMMA UNCHAINED LOGIC ---
    // ==========================================
    float _g_rsi_cap = _agresif_mode ? 99 : 90
    float _g_wick_tol = _agresif_mode ? 1.0 : 0.4
    
    bool _cond_safe = (close > _ema200) and (close > _ema50)
    bool _strong_candle = (_body_size > (_range_size * 0.5)) and (_upper_wick < (_body_size * _g_wick_tol))
    bool _rsi_valid = _rsi > 50 and _rsi < _g_rsi_cap
    
    bool _base_alpha = _cond_safe and _f_vol and (_dir == 1) and _strong_candle
    bool _final_alpha = _base_alpha and _rsi_valid
    bool _sig_gamma = _final_alpha and not _sig_vshape and not _sig_combo and not _sig_beta
    
    // Early Sweep
    bool _sweep_liq = not na(_last_swing_low) and (low < _last_swing_low)
    bool _rsi_div = not na(_last_swing_rsi) and (_rsi > _last_swing_rsi)
    bool _is_power_candle = (_body_size > (_range_size * 0.3)) and (_upper_wick < _body_size)
    bool _is_hammer = (_lower_wick > (_body_size * 2))
    bool _valid_candle = (close > open) and (_is_power_candle or _is_hammer)
    bool _early_trig = _sweep_liq and _rsi_div and _valid_candle and _valid_loc_sniper and _safe_from_crash
    bool _sig_early = _early_trig and not _sig_combo and not _sig_vshape

    string _stat_txt = _dir == 1 ? "UPTREND" : "DOWNTREND"
    color _stat_col = _dir == 1 ? c_up_fix : c_dn_fix
    
    string _sig_txt = "WAIT"
    color _sig_col = c_gray_fix
    
    if _sig_combo
        _sig_txt := "SNIPER COMBO"
        _sig_col := _col_snip
    else if _sig_beta
        _sig_txt := "BETA BREAKOUT"
        _sig_col := _col_beta
    else if _sig_early
        _sig_txt := "EARLY SWEEP"
        _sig_col := _col_earl
    else if _sig_vshape
        _sig_txt := "V-SHAPE"
        _sig_col := _col_vshp
    else if _sig_gamma
        _sig_txt := "GAMMA PUMP"
        _sig_col := _col_gamm
    else if _dir == 1
        _sig_txt := "HOLD"
        _sig_col := color.new(c_up_fix, 60)
    else
        _sig_txt := "AVOID"
        _sig_col := color.new(c_dn_fix, 60)
        
    float _chg = (close - close[1]) / close[1] * 100

    [_ema21, _ema50, _ema200, _stop_line, _dir, _sig_combo, _sig_gamma, _sig_vshape, _sig_early, _stat_txt, _stat_col, _sig_txt, _sig_col, close, _chg, _rsi, _dem_bot, _dem_top, adx_val, _sig_beta]

        else
            final_gamma := false

    int last_buy_ms = ta.valuewhen(any_buy, time, 0)
